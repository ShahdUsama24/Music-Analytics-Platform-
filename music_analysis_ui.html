# ── Query 9: Rank songs within each artist ──────────────────────
from pyspark.sql.window import Window

artist_window = (Window
    .partitionBy("artist")          # reset rank per artist
    .orderBy(F.desc("total_plays"))) # highest plays = rank 1

ranked = (clean_df
    .groupBy("artist", "song_name")
    .agg(F.sum("play_count").alias("total_plays"))
    .withColumn("rank_in_artist", F.rank().over(artist_window))
    .filter(F.col("rank_in_artist") == 1)   # keep only #1 per artist
    .orderBy(F.desc("total_plays"))
    .limit(10))

ranked.show(truncate=False)
